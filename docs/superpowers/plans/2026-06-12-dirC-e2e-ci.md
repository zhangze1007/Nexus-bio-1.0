import { defineConfig } from '@playwright/test';

export default defineConfig({
  testDir: './e2e',
  fullyParallel: true,
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 1 : undefined,
  reporter: 'html',
  use: {
    baseURL: 'http://localhost:3000',
    trace: 'on-first-retry',
  },
  webServer: {
    command: 'npm run dev',
    url: 'http://localhost:3000',
    reuseExistingServer: !process.env.CI,
  },
});

### Task C1.2: Write golden-path E2E test

**Files:**
- Create: `e2e/golden-path.spec.ts`

- [ ] **Step 1: Write the golden-path test**

```typescript
// e2e/golden-path.spec.ts
import { test, expect } from '@playwright/test';

test.describe('Golden Path: PATHD → FBASim → CatDes', () => {
  test('home page loads', async ({ page }) => {
    await page.goto('/');
    await expect(page.locator('text=Nexus-Bio')).toBeVisible();
  });

  test('PATHD tool loads and shows KEGG search', async ({ page }) => {
    await page.goto('/tools/pathd');
    await expect(page.locator('input[placeholder*="KEGG"]')).toBeVisible();
  });

  test('FBASim tool loads and shows model selector', async ({ page }) => {
    await page.goto('/tools/fbasim');
    await expect(page.locator('select')).toBeVisible();
  });

  test('CatDes tool loads', async ({ page }) => {
    await page.goto('/tools/catdes');
    await expect(page.locator('text=Enzyme')).toBeVisible();
  });

  test('CellFree tool loads', async ({ page }) => {
    await page.goto('/tools/cellfree');
    await expect(page.locator('text=Cell-Free')).toBeVisible();
  });

  test('DBTLflow tool loads', async ({ page }) => {
    await page.goto('/tools/dbtlflow');
    await expect(page.locator('text=DBTL')).toBeVisible();
  });
});
```

- [ ] **Step 2: Run E2E test**

Run: `npx playwright test --headed`
Expected: All tests pass (or skip if dev server not running)

## Phase C2: CI Hardening

### Task C2.1: Add E2E step to CI workflow

**Files:**
- Modify: `.github/workflows/ci.yml`

- [ ] **Step 1: Add E2E job**

```yaml
e2e:
  runs-on: ubuntu-latest
  needs: [test, typecheck]
  steps:
    - uses: actions/checkout@v4
    - uses: actions/setup-node@v4
      with:
        node-version: '20'
    - run: npm ci
    - run: npx playwright install --with-deps
    - run: npm run build
    - run: npx playwright test
    - uses: actions/upload-artifact@v4
      if: always()
      with:
        name: playwright-report
        path: playwright-report/
```

- [ ] **Step 2: Commit**

```bash
git add .github/workflows/ci.yml
git commit -m "ci: add E2E test step to GitHub Actions workflow"
```
